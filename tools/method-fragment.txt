    /**
     * 旷课预警生成/刷新（考勤服务 Feign 调用）
     * 幂等：存在未处理的旷课预警则更新，否则新建
     */
    @Override
    public void createTruantWarning(TruantWarningDTO dto) {
        if (dto == null || dto.getUserId() == null) {
            log.warn("旷课预警入参非法：{}", dto);
            return;
        }
        try {
            java.time.LocalDateTime now = java.time.LocalDateTime.now();
            // 幂等：存在未处理的旷课预警则更新，否则新建
            Warning existing = warningMapper.selectOne(
                    Wrappers.<Warning>lambdaQuery()
                            .eq(Warning::getUserId, dto.getUserId())
                            .eq(Warning::getWarningType, "absent")
                            .eq(Warning::getStatus, 1)
                            .last("LIMIT 1"));
            if (existing != null) {
                existing.setLevel(dto.getLevel());
                existing.setTitle(dto.getTitle());
                existing.setContent(dto.getContent());
                existing.setUpdateTime(now);
                warningMapper.updateById(existing);
            } else {
                Warning warning = new Warning();
                warning.setUserId(dto.getUserId());
                warning.setWarningType("absent");
                warning.setLevel(dto.getLevel());
                warning.setTitle(dto.getTitle());
                warning.setContent(dto.getContent());
                warning.setStatus(1);
                warning.setCreateTime(now);
                warning.setUpdateTime(now);
                warningMapper.insert(warning);
            }
            log.info("旷课预警已生成/刷新，studentId={}, 累计旷课={}次, level={}",
                    dto.getUserId(), dto.getTruantCount(), dto.getLevel());
        } catch (Exception e) {
            log.error("生成旷课预警失败，studentId={}", dto.getUserId(), e);
        }
    }
